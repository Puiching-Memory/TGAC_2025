### 成功案例 1
- **SQL ID**: `sql_78`
- **得分**: 1
- **来源**: CKPT `V7.1_22.09_1108`

**SQL:**

```sql
SELECT
    -- 初日华服回归抽奖 & 沙漠统治者有抽奖人数
    COUNT(DISTINCT CASE WHEN l815.vplayerid IS NOT NULL AND l690.vplayerid IS NOT NULL THEN l815.vplayerid END) AS 初日华服回归抽奖_沙漠统治者有抽奖人数,

    -- 初日华服回归抽奖 & 境界双花有抽奖人数
    COUNT(DISTINCT CASE WHEN l815.vplayerid IS NOT NULL AND l748.vplayerid IS NOT NULL THEN l815.vplayerid END) AS 初日华服回归抽奖_境界双花有抽奖人数,

    -- 初日华服回归抽奖 & 玄龙凌霄有抽奖人数
    COUNT(DISTINCT CASE WHEN l815.vplayerid IS NOT NULL AND l750.vplayerid IS NOT NULL THEN l815.vplayerid END) AS 初日华服回归抽奖_玄龙凌霄有抽奖人数,

    -- 初日华服回归抽奖 & 沙漠统治者有抽奖 & 但没有其主套人数
    COUNT(DISTINCT CASE 
        WHEN l815.vplayerid IS NOT NULL 
         AND l690.vplayerid IS NOT NULL 
         AND gear_430179.vplayerid IS NULL 
        THEN l815.vplayerid 
    END) AS 初日华服回归抽奖_沙漠统治者有抽奖但无主套人数,

    -- 初日华服回归抽奖 & 境界双花有抽奖 & 但没有其主套人数
    COUNT(DISTINCT CASE 
        WHEN l815.vplayerid IS NOT NULL 
         AND l748.vplayerid IS NOT NULL 
         AND gear_430265.vplayerid IS NULL 
        THEN l815.vplayerid 
    END) AS 初日华服回归抽奖_境界双花有抽奖但无主套人数,

    -- 初日华服回归抽奖 & 玄龙凌霄有抽奖 & 但没有其主套人数
    COUNT(DISTINCT CASE 
        WHEN l815.vplayerid IS NOT NULL 
         AND l750.vplayerid IS NOT NULL 
         AND gear_430225.vplayerid IS NULL 
        THEN l815.vplayerid 
    END) AS 初日华服回归抽奖_玄龙凌霄有抽奖但无主套人数,

    -- 初日华服回归抽奖 & 【沙漠统治者和境界双花有主套】人数
    COUNT(DISTINCT CASE 
        WHEN l815.vplayerid IS NOT NULL 
         AND gear_430179.vplayerid IS NOT NULL 
         AND gear_430265.vplayerid IS NOT NULL 
        THEN l815.vplayerid 
    END) AS 初日华服回归抽奖_沙漠统治者_境界双花有主套人数,

    -- 初日华服回归抽奖 & 【沙漠统治者和玄龙凌霄有主套】人数
    COUNT(DISTINCT CASE 
        WHEN l815.vplayerid IS NOT NULL 
         AND gear_430179.vplayerid IS NOT NULL 
         AND gear_430225.vplayerid IS NOT NULL 
        THEN l815.vplayerid 
    END) AS 初日华服回归抽奖_沙漠统治者_玄龙凌霄有主套人数,

    -- 初日华服回归抽奖 & 【境界双花和玄龙凌霄有主套】人数
    COUNT(DISTINCT CASE 
        WHEN l815.vplayerid IS NOT NULL 
         AND gear_430265.vplayerid IS NOT NULL 
         AND gear_430225.vplayerid IS NOT NULL 
        THEN l815.vplayerid 
    END) AS 初日华服回归抽奖_境界双花_玄龙凌霄有主套人数,

    -- 初日华服回归抽奖但【沙漠统治者&境界双花&玄龙凌霄】没有抽奖人数
    COUNT(DISTINCT CASE 
        WHEN l815.vplayerid IS NOT NULL 
         AND l690.vplayerid IS NULL 
         AND l748.vplayerid IS NULL 
         AND l750.vplayerid IS NULL 
        THEN l815.vplayerid 
    END) AS 初日华服回归抽奖_三套均未抽奖人数

FROM 
    (SELECT DISTINCT vplayerid FROM dwd_jordass_lotteryrecord_hi WHERE activitytype = 815 AND tdbank_imp_date LIKE '20250119%') l815

-- 关联其他抽奖活动
LEFT JOIN (SELECT DISTINCT vplayerid FROM dwd_jordass_lotteryrecord_hi WHERE activitytype = 690 AND tdbank_imp_date BETWEEN '20240426' AND '20240606') l690 ON l815.vplayerid = l690.vplayerid
LEFT JOIN (SELECT DISTINCT vplayerid FROM dwd_jordass_lotteryrecord_hi WHERE activitytype = 748 AND tdbank_imp_date BETWEEN '20240607' AND '20240725') l748 ON l815.vplayerid = l748.vplayerid
LEFT JOIN (SELECT DISTINCT vplayerid FROM dwd_jordass_lotteryrecord_hi WHERE activitytype = 750 AND tdbank_imp_date BETWEEN '20240726' AND '20240822') l750 ON l815.vplayerid = l750.vplayerid

-- 关联主套获取记录（addorreduce = 0 表示获得）
LEFT JOIN (SELECT DISTINCT vplayerid FROM dwd_jordass_gearlog_hi WHERE igoodsid = 430179 AND addorreduce = 0) gear_430179 ON l815.vplayerid = gear_430179.vplayerid
LEFT JOIN (SELECT DISTINCT vplayerid FROM dwd_jordass_gearlog_hi WHERE igoodsid = 430265 AND addorreduce = 0) gear_430265 ON l815.vplayerid = gear_430265.vplayerid
LEFT JOIN (SELECT DISTINCT vplayerid FROM dwd_jordass_gearlog_hi WHERE igoodsid = 430225 AND addorreduce = 0) gear_430225 ON l815.vplayerid = gear_430225.vplayerid
```
