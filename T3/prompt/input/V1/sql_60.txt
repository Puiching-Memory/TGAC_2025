WITH date_series AS (
    SELECT DISTINCT dtstatdate

alliance_users AS (
    SELECT DISTINCT l.vplayerid
    FROM dwd_jordass_player_allianceactive_hi a

SELECT 
    ds.dtstatdate,
    COUNT(DISTINCT w.vplayerid) AS paid_count
FROM date_series ds
JOIN dws_jordass_water_di w ON w.dtstatdate >= DATE_SUB(ds.dtstatdate, INTERVAL 29 DAY)
    AND w.dtstatdate <= ds.dtstatdate
