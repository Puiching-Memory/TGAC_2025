### vllm推理环境

CUDA 12.8.1 ubuntu 24 H800 * 8 sm90

```bash
apt update
apt upgrade
apt install git wget curl libxml2 build-essential htop nvtop iputils-ping

bash cuda_12.8.1_570.124.06_linux.run

curl -LsSf https://astral.sh/uv/install.sh | sh

wget https://gitee.com/RubyMetric/chsrc/releases/download/pre/chsrc_latest-1_amd64.deb
apt install ./chsrc_latest-1_amd64.deb
chsrc set uv

uv venv --python 3.13 --seed
source .venv/bin/activate
uv pip install vllm --torch-backend=auto

hf download XGenerationLab/XiYanSQL-QwenCoder-7B-2504 --repo-type model

export CUDA_VISIBLE_DEVICES=3
vllm serve Qwen/Qwen3-0.6B # 小模型测试
vllm serve Qwen/Qwen3-4B-Thinking-2507 # 糟糕的指令遵从性
vllm serve Qwen/Qwen3-Coder-30B-A3B-Instruct-FP8 # vllm 目前不支持
vllm serve XGenerationLab/XiYanSQL-QwenCoder-7B-2504 \
           --dtype auto \
           --gpu-memory-utilization 0.2 \
           --max-model-len 8192 \
           --port 8891 \
           --host 0.0.0.0 \
           --data-parallel-size 2 \
           --api-key "" \
           --enable-auto-tool-choice \
           --tool-call-parser hermes
```
